/* =====================================================================
   KINET - Script de creación de base de datos (Oracle 12c+ / 19c / 21c / XE)
   Modelo: MER lógico normalizado 3FN
   Ejecutar como SYS / SYSDBA, conectado al PDB (ej: ALTER SESSION SET CONTAINER = XEPDB1;)
   Secciones:
     1. Tablespaces
     2. Perfil, usuarios y roles
     3. Tablas (esquema KINET_OWNER)
     4. Índices
     5. Triggers, vistas
     6. Datos semilla
     7. Permisos y sinónimos
   IMPORTANTE: cambiar las contraseñas antes de ejecutar.
   ===================================================================== */

SET DEFINE OFF
SET SQLBLANKLINES ON
SET SERVEROUTPUT ON
WHENEVER SQLERROR CONTINUE

/* =====================================================================
   0. LIMPIEZA (permite re-ejecutar el script completo desde cero)
   ===================================================================== */
BEGIN
  FOR u IN (SELECT username FROM dba_users
            WHERE username IN ('KINET_OWNER','KINET_APP','KINET_READER','KINET_ADMIN')) LOOP
    FOR s IN (SELECT sid, serial# FROM v$session WHERE username = u.username) LOOP
      BEGIN
        EXECUTE IMMEDIATE 'ALTER SYSTEM KILL SESSION ''' || s.sid || ',' || s.serial# || ''' IMMEDIATE';
      EXCEPTION WHEN OTHERS THEN NULL;
      END;
    END LOOP;
    EXECUTE IMMEDIATE 'DROP USER ' || u.username || ' CASCADE';
  END LOOP;
  FOR r IN (SELECT role FROM dba_roles
            WHERE role IN ('ROL_KINET_OWNER','ROL_KINET_APP','ROL_KINET_READER','ROL_KINET_ADMIN')) LOOP
    EXECUTE IMMEDIATE 'DROP ROLE ' || r.role;
  END LOOP;
  FOR p IN (SELECT profile FROM dba_profiles WHERE profile = 'KINET_PROFILE' AND ROWNUM = 1) LOOP
    EXECUTE IMMEDIATE 'DROP PROFILE KINET_PROFILE CASCADE';
  END LOOP;
  FOR t IN (SELECT tablespace_name FROM dba_tablespaces
            WHERE tablespace_name IN ('KINET_DATA','KINET_IDX')) LOOP
    EXECUTE IMMEDIATE 'DROP TABLESPACE ' || t.tablespace_name || ' INCLUDING CONTENTS AND DATAFILES';
  END LOOP;
END;
/

/* =====================================================================
   1. TABLESPACES
   (Usa Oracle Managed Files; si db_create_file_dest no está definido,
    agregar la ruta: DATAFILE '/ruta/kinet_data01.dbf' SIZE 200M ...)
   ===================================================================== */
CREATE TABLESPACE kinet_data
  DATAFILE SIZE 200M AUTOEXTEND ON NEXT 50M MAXSIZE UNLIMITED
  EXTENT MANAGEMENT LOCAL SEGMENT SPACE MANAGEMENT AUTO;

CREATE TABLESPACE kinet_idx
  DATAFILE SIZE 100M AUTOEXTEND ON NEXT 25M MAXSIZE UNLIMITED
  EXTENT MANAGEMENT LOCAL SEGMENT SPACE MANAGEMENT AUTO;

/* =====================================================================
   2. PERFIL, USUARIOS Y ROLES
   ===================================================================== */
CREATE PROFILE kinet_profile LIMIT
  FAILED_LOGIN_ATTEMPTS   5
  PASSWORD_LOCK_TIME      1/24
  PASSWORD_LIFE_TIME      180
  PASSWORD_REUSE_TIME     365
  PASSWORD_REUSE_MAX      5
  SESSIONS_PER_USER       20
  IDLE_TIME               60;

-- Dueño del esquema (no se usa desde la aplicación)
CREATE USER kinet_owner IDENTIFIED BY "Cambiar_Owner#2026"
  DEFAULT TABLESPACE kinet_data
  TEMPORARY TABLESPACE temp
  QUOTA UNLIMITED ON kinet_data
  QUOTA UNLIMITED ON kinet_idx
  PROFILE kinet_profile;

-- Usuario de la aplicación (backend)
CREATE USER kinet_app IDENTIFIED BY "Cambiar_App#2026"
  DEFAULT TABLESPACE kinet_data
  TEMPORARY TABLESPACE temp
  PROFILE kinet_profile;

-- Usuario de solo lectura (reportes / BI)
CREATE USER kinet_reader IDENTIFIED BY "Cambiar_Read#2026"
  DEFAULT TABLESPACE kinet_data
  TEMPORARY TABLESPACE temp
  PROFILE kinet_profile;

-- Usuario administrador de la BD KINET (DBA funcional)
CREATE USER kinet_admin IDENTIFIED BY "Cambiar_Admin#2026"
  DEFAULT TABLESPACE kinet_data
  TEMPORARY TABLESPACE temp
  QUOTA UNLIMITED ON kinet_data
  PROFILE kinet_profile;

-- Roles
CREATE ROLE rol_kinet_owner;
CREATE ROLE rol_kinet_app;
CREATE ROLE rol_kinet_reader;
CREATE ROLE rol_kinet_admin;

-- Privilegios de sistema
GRANT CREATE SESSION, CREATE TABLE, CREATE VIEW, CREATE SEQUENCE,
      CREATE TRIGGER, CREATE PROCEDURE, CREATE SYNONYM, CREATE TYPE
  TO rol_kinet_owner;
GRANT CREATE SESSION TO rol_kinet_app;
GRANT CREATE SESSION TO rol_kinet_reader;
GRANT CREATE SESSION, CREATE SYNONYM TO rol_kinet_admin;

GRANT rol_kinet_owner  TO kinet_owner;
GRANT rol_kinet_app    TO kinet_app;
GRANT rol_kinet_reader TO kinet_reader;
GRANT rol_kinet_admin  TO kinet_admin;

-- Las tablas se crean en el esquema KINET_OWNER desde esta sesión
ALTER SESSION SET CURRENT_SCHEMA = kinet_owner;

/* ---------- USUARIOS Y AUTENTICACIÓN ---------- */

  id_usuario        NUMBER GENERATED ALWAYS AS IDENTITY,

  correo            VARCHAR2(150)  NOT NULL,   -- UQ

  estado            VARCHAR2(20) DEFAULT 'ACTIVO' NOT NULL,
  foto_perfil       VARCHAR2(500),   -- ? opcional (NULL)
  CONSTRAINT pk_usuario PRIMARY KEY (id_usuario) USING INDEX TABLESPACE kinet_idx,
  CONSTRAINT uq_usuario_correo UNIQUE (correo) USING INDEX TABLESPACE kinet_idx,   -- UQ (MER)
  CONSTRAINT ck_usuario_estado CHECK (estado IN ('ACTIVO','INACTIVO','SUSPENDIDO','ELIMINADO')),
  CONSTRAINT ck_usuario_correo CHECK (INSTR(correo,'@') > 1)

  id_usuario        NUMBER NOT NULL,
  id_entrenamiento  NUMBER,   -- ? opcional (NULL)
  id_usuario_mision NUMBER,   -- ? opcional (NULL)
  id_usuario_reto   NUMBER,   -- ? opcional (NULL)
  cantidad_xp       NUMBER(10) NOT NULL,      -- (+/-)
  motivo            VARCHAR2(255) NOT NULL,
  fecha_registro    TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT pk_movimiento_xp PRIMARY KEY (id_movimiento_xp) USING INDEX TABLESPACE kinet_idx,
  CONSTRAINT fk_movxp_usuario FOREIGN KEY (id_usuario)        REFERENCES usuario (id_usuario),
  CONSTRAINT fk_movxp_entren  FOREIGN KEY (id_entrenamiento)  REFERENCES entrenamiento (id_entrenamiento),
  CONSTRAINT fk_movxp_usmis   FOREIGN KEY (id_usuario_mision) REFERENCES usuario_mision (id_usuario_mision),
  CONSTRAINT fk_movxp_usret   FOREIGN KEY (id_usuario_reto)   REFERENCES usuario_reto (id_usuario_reto),
  CONSTRAINT ck_movxp_cant    CHECK (cantidad_xp <> 0),
  -- a lo sumo un origen por movimiento
  CONSTRAINT ck_movxp_origen  CHECK (
    (CASE WHEN id_entrenamiento  IS NOT NULL THEN 1 ELSE 0 END +
     CASE WHEN id_usuario_mision IS NOT NULL THEN 1 ELSE 0 END +
     CASE WHEN id_usuario_reto   IS NOT NULL THEN 1 ELSE 0 END) <= 1)
) TABLESPACE kinet_data;

-- Mantiene la mejor racha y el límite de dias_actuales
CREATE OR REPLACE TRIGGER trg_racha_mejor
BEFORE INSERT OR UPDATE ON racha
FOR EACH ROW
BEGIN
  IF :NEW.dias_actuales > :NEW.mejor_racha THEN
    :NEW.mejor_racha := :NEW.dias_actuales;
  END IF;
END;
/

-- Registra el historial de cambio de rango automáticamente
CREATE OR REPLACE TRIGGER trg_hist_rango
AFTER UPDATE OF id_rango ON participacion_temporada
FOR EACH ROW
WHEN (OLD.id_rango <> NEW.id_rango)
BEGIN
  INSERT INTO historial_rango (id_participacion, id_rango_anterior, id_rango_nuevo, motivo)
  VALUES (:NEW.id_participacion, :OLD.id_rango, :NEW.id_rango, 'Cambio de rango por XP de temporada');
END;
/

-- Vista pública de usuario (sin password_hash) para reportes
CREATE OR REPLACE VIEW v_usuario_publico AS
SELECT id_usuario, nombre, apellido, fecha_registro, estado, foto_perfil
FROM usuario;

-- XP total por usuario
CREATE OR REPLACE VIEW v_xp_total_usuario AS
SELECT u.id_usuario, u.nombre, u.apellido,
       NVL(SUM(m.cantidad_xp), 0) AS xp_total
FROM usuario u
LEFT JOIN movimiento_xp m ON m.id_usuario = u.id_usuario
GROUP BY u.id_usuario, u.nombre, u.apellido;

-- Ranking de una temporada
CREATE OR REPLACE VIEW v_ranking_temporada AS
SELECT pt.id_temporada, pt.id_usuario, u.nombre, u.apellido, r.nombre AS rango,
       pt.xp_temporada,
       RANK() OVER (PARTITION BY pt.id_temporada ORDER BY pt.xp_temporada DESC) AS posicion
FROM participacion_temporada pt
JOIN usuario u ON u.id_usuario = pt.id_usuario
JOIN rango   r ON r.id_rango   = pt.id_rango
WHERE pt.estado = 'ACTIVA';

  END LOOP;

       LISTAGG(cc.column_name, ', ') WITHIN GROUP (ORDER BY cc.position) AS columnas

GROUP BY c.table_name, c.constraint_name
ORDER BY 1, 2;
